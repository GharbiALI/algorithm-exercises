var
  i, maxIndex: integer;

  maxIndex := 0;
  for i := 1 to numUsers - 1 do
  begin

    begin
      maxIndex := i;

    begin
      if CompareDates(users[i].lastLoginDate, users[maxIndex].lastLoginDate) > 0 then
