    queryStart, queryEnd, ampersandPos,equalPos : integer;

    queryStart := Pos('?', url);
    queryEnd := Pos('#', url);
    
    if queryStart = 0 then 
    begin
        writeln('No parameters found in URL');
    end
    else
    begin
        queryString := Copy(url, queryStart + 1, queryEnd - queryStart - 1) + '&';
        writeln(queryString);
        
        writeln('------------- URL Parameters : ------------------');
        while(queryString <> '')do
        begin 
            equalPos := Pos('=', queryString);
            ampersandPos := Pos('&', queryString);
            
            param := Copy(queryString, 1, equalPos - 1);
            val := Copy(queryString, equalPos + 1, ampersandPos - equalPos - 1);
            if val = '' then
            begin
                val := 'null';
            end;
            
            writeln(param,'--> ',val);
            Delete(queryString, 1, ampersandPos);
            
            
        end;
        
        
    
    end;
end.
